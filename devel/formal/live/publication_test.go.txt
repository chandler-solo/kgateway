type liveTrace struct {
	mu     sync.Mutex
	start  time.Time
	events []map[string]any
}

func (tr *liveTrace) add(kind string, fields map[string]any) {
	tr.mu.Lock()
	defer tr.mu.Unlock()
	fields["schema"], fields["seq"], fields["kind"], fields["elapsed_ns"] = 1, len(tr.events)+1, kind, time.Since(tr.start).Nanoseconds()
	tr.events = append(tr.events, fields)
}

func liveHash(data []byte) string { sum := sha256.Sum256(data); return hex.EncodeToString(sum[:]) }

type liveCache struct {
	cache.SnapshotCache
	trace      *liveTrace
	controller string
}

func (c *liveCache) SetSnapshot(ctx context.Context, key string, snap cache.ResourceSnapshot) error {
	versions, hashes := map[string]string{}, map[string]map[string]string{}
	for _, typ := range []string{resource.ClusterType, resource.EndpointType, resource.ListenerType, resource.RouteType, resource.SecretType} {
		versions[typ], hashes[typ] = snap.GetVersion(typ), map[string]string{}
		for name, item := range snap.GetResources(typ) {
			data, err := (proto.MarshalOptions{Deterministic: true}).Marshal(item)
			if err != nil {
				return err
			}
			hashes[typ][name] = liveHash(data)
		}
	}
	c.trace.add("install-attempt", map[string]any{"controller": c.controller, "key": key, "versions": versions, "resources": hashes})
	err := c.SnapshotCache.SetSnapshot(ctx, key, snap)
	installed, readErr := c.SnapshotCache.GetSnapshot(key)
	status := "ok"
	if err != nil {
		status = err.Error()
	}
	c.trace.add("install-result", map[string]any{"controller": c.controller, "key": key, "version": snap.GetVersion(resource.RouteType), "status": status, "readback_matches": readErr == nil && installed == snap})
	return err
}

type liveRelay struct {
	discovery.UnimplementedAggregatedDiscoveryServiceServer
	mu         sync.Mutex
	trace      *liveTrace
	target     string
	addresses  map[string]string
	generation int
	cancel     context.CancelFunc
	wg         sync.WaitGroup
}

func (p *liveRelay) handoff(target string) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.target = target
	p.trace.add("handoff", map[string]any{"target": target})
	if p.cancel != nil {
		p.cancel()
	}
}

func (p *liveRelay) StreamAggregatedResources(front discovery.AggregatedDiscoveryService_StreamAggregatedResourcesServer) error {
	p.wg.Add(1)
	defer p.wg.Done()
	p.mu.Lock()
	p.generation++
	gen, target := p.generation, p.target
	ctx, cancel := context.WithCancel(front.Context())
	p.cancel = cancel
	p.trace.add("stream-open", map[string]any{"controller": target, "generation": gen})
	p.mu.Unlock()
	defer cancel()
	defer p.trace.add("stream-close", map[string]any{"controller": target, "generation": gen})
	conn, err := grpc.NewClient(p.addresses[target], grpc.WithTransportCredentials(insecure.NewCredentials()))
	if err != nil {
		return err
	}
	defer conn.Close()
	md, _ := metadata.FromIncomingContext(front.Context())
	up, err := discovery.NewAggregatedDiscoveryServiceClient(conn).StreamAggregatedResources(metadata.NewOutgoingContext(ctx, md.Copy()), grpc.WaitForReady(true))
	if err != nil {
		return err
	}
	p.wg.Go(func() {
		for {
			r, e := front.Recv()
			if e != nil {
				cancel()
				return
			}
			p.trace.add("request", map[string]any{"controller": target, "generation": gen, "type": r.TypeUrl, "version": r.VersionInfo, "nonce": r.ResponseNonce, "names": r.ResourceNames, "nack": r.ErrorDetail != nil})
			if up.Send(r) != nil {
				cancel()
				return
			}
		}
	})
	for {
		r, e := up.Recv()
		if e != nil {
			return e
		}
		hashes := []string{}
		for _, item := range r.Resources {
			hashes = append(hashes, liveHash(item.Value))
		}
		p.trace.add("send-attempt", map[string]any{"controller": target, "generation": gen, "type": r.TypeUrl, "version": r.VersionInfo, "nonce": r.Nonce, "resources": hashes})
		e = front.Send(r)
		status := "ok"
		if e != nil {
			status = e.Error()
		}
		p.trace.add("send-result", map[string]any{"controller": target, "generation": gen, "type": r.TypeUrl, "version": r.VersionInfo, "nonce": r.Nonce, "status": status})
		if e != nil {
			return e
		}
	}
}

func liveBuild(t *testing.T, version string, missingCluster, missingEndpoints bool) proxy_syncer.FormalLiveBuild {
	t.Helper()
	ads := &envoycorev3.ConfigSource{ResourceApiVersion: envoycorev3.ApiVersion_V3, ConfigSourceSpecifier: &envoycorev3.ConfigSource_Ads{Ads: &envoycorev3.AggregatedConfigSource{}}}
	c := &envoyclusterv3.Cluster{Name: "a", ConnectTimeout: durationpb.New(time.Second), ClusterDiscoveryType: &envoyclusterv3.Cluster_Type{Type: envoyclusterv3.Cluster_EDS}, EdsClusterConfig: &envoyclusterv3.Cluster_EdsClusterConfig{EdsConfig: ads}}
	routes := &envoyroutev3.RouteConfiguration{Name: "routes", VirtualHosts: []*envoyroutev3.VirtualHost{{Name: "all", Domains: []string{"*"}, Routes: []*envoyroutev3.Route{
		{Match: &envoyroutev3.RouteMatch{PathSpecifier: &envoyroutev3.RouteMatch_Prefix{Prefix: "/backend"}}, Action: &envoyroutev3.Route_Route{Route: &envoyroutev3.RouteAction{ClusterSpecifier: &envoyroutev3.RouteAction_Cluster{Cluster: "a"}}}},
		{Match: &envoyroutev3.RouteMatch{PathSpecifier: &envoyroutev3.RouteMatch_Prefix{Prefix: "/"}}, Action: &envoyroutev3.Route_DirectResponse{DirectResponse: &envoyroutev3.DirectResponseAction{Status: 200, Body: &envoycorev3.DataSource{Specifier: &envoycorev3.DataSource_InlineString{InlineString: version}}}}},
	}}}}
	hcm := &hcmv3.HttpConnectionManager{StatPrefix: "formal-live", RouteSpecifier: &hcmv3.HttpConnectionManager_Rds{Rds: &hcmv3.Rds{RouteConfigName: "routes", ConfigSource: ads}}, HttpFilters: []*hcmv3.HttpFilter{{Name: "envoy.filters.http.router", ConfigType: &hcmv3.HttpFilter_TypedConfig{TypedConfig: livePacked(t, &routerv3.Router{})}}}}
	l := &envoylistenerv3.Listener{Name: "front", Address: &envoycorev3.Address{Address: &envoycorev3.Address_SocketAddress{SocketAddress: &envoycorev3.SocketAddress{Address: "0.0.0.0", PortSpecifier: &envoycorev3.SocketAddress_PortValue{PortValue: 10000}}}}, FilterChains: []*envoylistenerv3.FilterChain{{Filters: []*envoylistenerv3.Filter{{Name: "envoy.filters.network.http_connection_manager", ConfigType: &envoylistenerv3.Filter_TypedConfig{TypedConfig: livePacked(t, hcm)}}}}}}
	cla := &envoyendpointv3.ClusterLoadAssignment{ClusterName: "a"}
	if !missingEndpoints {
		cla.Endpoints = []*envoyendpointv3.LocalityLbEndpoints{{LbEndpoints: []*envoyendpointv3.LbEndpoint{{HostIdentifier: &envoyendpointv3.LbEndpoint_Endpoint{Endpoint: &envoyendpointv3.Endpoint{Address: &envoycorev3.Address{Address: &envoycorev3.Address_SocketAddress{SocketAddress: &envoycorev3.SocketAddress{Address: "127.0.0.1", PortSpecifier: &envoycorev3.SocketAddress_PortValue{PortValue: 10001}}}}}}}}}}
	}
	snap := &cache.Snapshot{}
	clusters, endpoints := []cachetypes.Resource{c}, []cachetypes.Resource{cla}
	if missingCluster {
		clusters, endpoints = nil, nil
	}
	cdsVersion := "cds-a"
	if missingCluster {
		cdsVersion = "cds-missing"
	}
	snap.Resources[cachetypes.Cluster] = cache.NewResources(cdsVersion, clusters)
	// The bridge applies the production version constructor after classifying
	// ready versus synthesized/absent assignments; the ready contribution is 17.
	snap.Resources[cachetypes.Endpoint] = cache.NewResources("17", endpoints)
	snap.Resources[cachetypes.Listener] = cache.NewResources("lds-front", []cachetypes.Resource{l})
	snap.Resources[cachetypes.Route] = cache.NewResources(version, []cachetypes.Resource{routes})
	snap.Resources[cachetypes.Secret] = cache.NewResources("sds-none", nil)
	w := proxy_syncer.FormalLiveBuild{Snapshot: snap}
	if missingCluster {
		w.MissingClusters = []string{"a"}
	}
	if missingEndpoints {
		w.MissingEndpoints = []string{"a"}
	}
	return w
}

func liveDocker(t *testing.T, args ...string) string {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 45*time.Second)
	defer cancel()
	b, e := exec.CommandContext(ctx, "docker", args...).CombinedOutput()
	require.NoError(t, e, "docker %v: %s", args, b)
	return strings.TrimSpace(string(b))
}

func liveGet(url string) (int, []byte, error) {
	c := &http.Client{Timeout: time.Second, Transport: &http.Transport{DisableKeepAlives: true}}
	r, e := c.Get(url)
	if e != nil {
		return 0, nil, e
	}
	defer r.Body.Close()
	b, e := io.ReadAll(r.Body)
	return r.StatusCode, b, e
}

func liveRouteVersion(body []byte) string {
	var dump struct {
		Configs []struct {
			Routes []struct {
				Version string `json:"version_info"`
				Route   struct {
					Name string `json:"name"`
				} `json:"route_config"`
			} `json:"dynamic_route_configs"`
		} `json:"configs"`
	}
	if json.Unmarshal(body, &dump) != nil {
		return ""
	}
	for _, c := range dump.Configs {
		for _, r := range c.Routes {
			if r.Route.Name == "routes" {
				return r.Version
			}
		}
	}
	return ""
}

func liveClusterState(body []byte) (bool, bool) {
	var dump struct {
		Configs []struct {
			Active []struct {
				Cluster struct {
					Name string `json:"name"`
				} `json:"cluster"`
			} `json:"dynamic_active_clusters"`
			Warming []struct {
				Cluster struct {
					Name string `json:"name"`
				} `json:"cluster"`
			} `json:"dynamic_warming_clusters"`
		} `json:"configs"`
	}
	if json.Unmarshal(body, &dump) != nil {
		return false, false
	}
	active, warming := false, false
	for _, cfg := range dump.Configs {
		for _, c := range cfg.Active {
			active = active || c.Cluster.Name == "a"
		}
		for _, c := range cfg.Warming {
			warming = warming || c.Cluster.Name == "a"
		}
	}
	return active, warming
}

func liveWait(t *testing.T, ch <-chan struct{}, what string) {
	t.Helper()
	select {
	case <-ch:
	case <-time.After(20 * time.Second):
		t.Fatal(what)
	}
}

func (f *liveFixture) observe(phase, version string, backend int) {
	f.t.Helper()
	var dump, before, after, back []byte
	require.Eventually(f.t, func() bool {
		code, b, e := liveGet(f.front)
		if e != nil || code != 200 || string(b) != version {
			return false
		}
		before = b
		code, d, e := liveGet(f.admin + "/config_dump?include_eds")
		if e != nil || code != 200 || liveRouteVersion(d) != version {
			return false
		}
		active, warming := liveClusterState(d)
		if (backend == 200 && (!active || warming)) || (backend == 503 && active) {
			return false
		}
		dump = d
		code, b, e = liveGet(f.front + "/backend")
		if e != nil || code != backend {
			return false
		}
		back = b
		code, after, e = liveGet(f.front)
		return e == nil && code == 200 && string(after) == version
	}, 25*time.Second, 20*time.Millisecond, "independent admin and traffic phase %s", phase)
	file := phase + "-config.json"
	require.NoError(f.t, os.WriteFile(filepath.Join(f.dir, file), dump, 0o600))
	f.trace.add("observation", map[string]any{"phase": phase, "version": version, "config": file, "sha256": liveHash(dump), "before": string(before), "after": string(after), "backend_code": backend, "backend_body": string(back)})
}

func (f *liveFixture) toA() {
	f.relay.handoff("A")
	require.Eventually(f.t, func() bool { return f.a.Warm() && f.b.Disconnected() }, 20*time.Second, time.Millisecond, "real callbacks classify returning Envoy as warm and close B")
	f.trace.add("warm-confirmed", map[string]any{"controller": "A", "key": f.key})
}

func (f *liveFixture) queued() {
	b := f.a.Pause("queued-old")
	defer b.Release()
	defer b.ReleaseTail()
	f.a.Put(liveBuild(f.t, "queued-old", false, false))
	liveWait(f.t, b.Arrived, "old event queued")
	f.a.Put(liveBuild(f.t, "candidate-d", false, true))
	f.toA()
	require.Equal(f.t, "candidate-d", f.a.State()["pending"])
	f.trace.add("pending-confirmed", f.a.State())
	b.Release()
	liveWait(f.t, b.Done, "old event attempted publication")
	if f.negative {
		f.observe("rollback", "queued-old", 200)
	} else {
		f.observe("natural-budget", "candidate-d", 200)
	}
	b.ReleaseTail()
	f.a.Put(liveBuild(f.t, "recovered", false, false))
	liveWait(f.t, f.a.Completed("recovered"), "coherent recovery")
	f.observe("recovered", "recovered", 200)
}

func (f *liveFixture) carry() {
	f.toA()
	f.observe("connected-a", "old-a", 200)
	f.a.Put(liveBuild(f.t, "carry-first", true, false))
	liveWait(f.t, f.a.Completed("carry-first"), "first carry")
	f.observe("carry-first", "carry-first", 200)
	first := time.Now()
	time.Sleep(liveCarryBound / 2)
	f.a.Put(liveBuild(f.t, "coherent", false, false))
	liveWait(f.t, f.a.Completed("coherent"), "coherent recovery")
	f.observe("coherent", "coherent", 200)
	remaining := liveCarryBound + 200*time.Millisecond - time.Since(first)
	if remaining > 0 {
		time.Sleep(remaining)
	}
	f.a.Put(liveBuild(f.t, "carry-second", true, false))
	liveWait(f.t, f.a.Completed("carry-second"), "second carry")
	f.trace.add("second-carry", f.a.State())
	if f.negative {
		f.observe("premature-drop", "carry-second", 503)
	} else {
		f.observe("fresh-carry", "carry-second", 200)
		time.Sleep(200 * time.Millisecond)
		f.observe("fresh-carry-stable", "carry-second", 200)
		f.observe("natural-carry-expiry", "carry-second", 503)
	}
	f.a.Put(liveBuild(f.t, "recovered", false, false))
	liveWait(f.t, f.a.Completed("recovered"), "final recovery")
	f.observe("recovered", "recovered", 200)
}

func (f *liveFixture) reset() {
	b := f.a.Pause("queued-old")
	defer b.Release()
	defer b.ReleaseTail()
	f.a.Put(liveBuild(f.t, "queued-old", false, true))
	liveWait(f.t, b.Arrived, "queued build")
	f.a.Put(liveBuild(f.t, "candidate-d", false, true))
	reset, replay := f.a.PauseReset(), f.a.PauseReplay()
	defer reset.Release()
	defer replay.Release()
	f.relay.handoff("A")
	liveWait(f.t, reset.Arrived, "reset entered")
	b.Release()
	select {
	case <-b.Done:
		f.t.Fatal("build completed inside reset/adoption critical section")
	case <-time.After(100 * time.Millisecond):
	}
	f.trace.add("reset-excluded", map[string]any{})
	reset.Release()
	liveWait(f.t, replay.Arrived, "admission replay selected")
	f.a.Put(liveBuild(f.t, "recovered", false, false))
	replay.Release()
	liveWait(f.t, b.Done, "queued event rejected")
	require.Eventually(f.t, f.a.Warm, 20*time.Second, time.Millisecond)
	f.trace.add("warm-confirmed", map[string]any{"controller": "A", "key": f.key})
	time.Sleep(liveBudget + 100*time.Millisecond)
	f.observe("no-leaked-timer", "accepted-b", 200)
	b.ReleaseTail()
	liveWait(f.t, f.a.Completed("recovered"), "current queued event delivered")
	f.observe("recovered", "recovered", 200)
}

func (f *liveFixture) cancelPending() {
	b := f.a.Pause("queued-old")
	defer b.Release()
	defer b.ReleaseTail()
	f.a.Put(liveBuild(f.t, "queued-old", false, true))
	liveWait(f.t, b.Arrived, "queued event")
	f.a.Put(liveBuild(f.t, "candidate-d", false, true))
	f.toA()
	require.Equal(f.t, "candidate-d", f.a.State()["pending"])
	stale := f.a.CapturePending()
	f.trace.add("captured-undrained", map[string]any{"version": "candidate-d"})
	f.a.Put(liveBuild(f.t, "recovered", false, false))
	b.Release()
	b.ReleaseTail()
	liveWait(f.t, f.a.Completed("recovered"), "coherent publish cancels pending timer")
	f.observe("cancelled-before-expiry", "recovered", 200)
	f.trace.add("invoke-cancelled", map[string]any{"version": "candidate-d"})
	stale()
	time.Sleep(liveBudget + 100*time.Millisecond)
	f.observe("after-cancelled-callback", "recovered", 200)
}

func TestFormalLivePublication(t *testing.T) {
	for _, scenario := range []string{"queued", "carry", "reset", "cancel"} {
		t.Run(scenario, func(t *testing.T) {
			f := newLiveFixture(t, scenario)
			switch scenario {
			case "queued":
				f.queued()
			case "carry":
				f.carry()
			case "reset":
				f.reset()
			case "cancel":
				f.cancelPending()
			}
		})
	}
}
